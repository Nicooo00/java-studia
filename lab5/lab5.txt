        for(int i = 0; i < n; i++)
        {
            System.out.print("Podaj liczbe: ");
            liczby[i] = scanner.nextInt();
        }

        System.out.println("Utworzona macierz: ");

        for(int i = 0; i < n; i++)
        {  
            System.out.printf("%4d", liczby[i]);
        }

        System.out.println();

        int najmniejsza = liczby[0];
        int najwieksza = liczby[0];
        
        for(int i = 0; i < n-1; i++)
        {
            if(najmniejsza > liczby[i+1])
                {
                    najmniejsza = liczby[i+1];
                }
            else if(najwieksza < liczby[i+1])
                {
                    najwieksza = liczby[i+1];
                }
        }

        System.out.println("Najwieksza liczba: " + najwieksza);
        System.out.println("Najmniejsza liczba: " + najmniejsza);

class Algorithm
{
    int indexMin(int[] tablica)
    {
        int min_index = 0;
        for(int i = 0; i < tablica.length-1; i++)
        {
            if(tablica[min_index] > tablica[i+1])
            {
                min_index = i + 1;
            }
        }
        return min_index;
    }

    double average(int[] tablica)
    {
        double srednia = 0;
        double suma = 0;

        for(int i = 0; i < tablica.length; i++)
        {
            suma += tablica[i];
        }

        srednia = suma / tablica.length;

        return srednia;
    }
}

    {

        {
            for(int i = j; i >= 1; i--)
            {
                if(tablica[i] < tablica[i-1])
                {

                    tablica[i] = tablica[i-1];
                    tablica[i-1] = pomocnicza;

package lab5;

public class Zadanie11 {
        public static void sortuj(int[] tab) {

        for (int i = 1; i < tab.length; i++) {

            int liczba = tab[i];
            int j = i - 1;

            while (j >= 0 && tab[j] > liczba) {
                tab[j + 1] = tab[j];
                j--;
            }

            tab[j + 1] = liczba;
        }
    }

    public static void main(String[] args) {

        int[] tab = {5, 2, 8, 1, 3};

        sortuj(tab);

        for (int liczba : tab) {
            System.out.print(liczba + " ");
        }
    }
    
}
