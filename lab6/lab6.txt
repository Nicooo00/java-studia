package lab6;

import java.util.Arrays;
import java.text.Collator;
import java.util.Locale;

class Dane2 implements Comparable<Dane2> {
    String nazwisko;

    @Override
    public int compareTo(Dane2 inna) {
        Collator collator = Collator.getInstance(Locale.forLanguageTag("pl-PL"));

        return collator.compare(this.nazwisko, inna.nazwisko);
    }
}

public class Zadanie2 {
    public static void main(String[] args)
    {
        Dane2[] informacje = new Dane2[3];

        for(int i = 0; i < informacje.length; i++)
        {
            informacje[i] = new Dane2();
        }

        informacje[0].nazwisko = "Kowalski";
        informacje[1].nazwisko = "Nowak";
        informacje[2].nazwisko = "Adamski";

        System.out.println();

        for(int i = 0; i < informacje.length; i++)
        {
            System.out.println("Nazwisko: " + informacje[i].nazwisko);
        }

        Arrays.sort(informacje);

        for(Dane2 informacja : informacje)
        {
            System.out.println(informacja.nazwisko);
        }
    }
}

package lab6;

import java.util.Arrays;
import java.text.Collator;
import java.util.Locale;

class Pracownik implements Comparable<Pracownik> {
    int id;
    String nazwisko;
    String stanowisko;

    @Override
    public int compareTo(Pracownik inny) {
        Collator collator = Collator.getInstance(Locale.forLanguageTag("pl-PL"));

        int wynik = collator.compare(this.stanowisko, inny.stanowisko);

        if (wynik == 0) {
            return collator.compare(this.nazwisko, inny.nazwisko);
        }

        return wynik;
    }
}

public class Zadanie4 {
    public static void main(String[] args) {

        Pracownik[] pracownicy = new Pracownik[5];

        for (int i = 0; i < pracownicy.length; i++)
        {
            pracownicy[i] = new Pracownik();
        }

        pracownicy[0].id = 1;
        pracownicy[0].nazwisko = "Kowalski";
        pracownicy[0].stanowisko = "Programista";

        pracownicy[1].id = 2;
        pracownicy[1].nazwisko = "Nowak";
        pracownicy[1].stanowisko = "Kierownik";

        pracownicy[2].id = 3;
        pracownicy[2].nazwisko = "Adamski";
        pracownicy[2].stanowisko = "Programista";

        pracownicy[3].id = 4;
        pracownicy[3].nazwisko = "Wisniewski";
        pracownicy[3].stanowisko = "Analityk";

        pracownicy[4].id = 5;
        pracownicy[4].nazwisko = "Baran";
        pracownicy[4].stanowisko = "Kierownik";

        Arrays.sort(pracownicy);

        for (Pracownik p : pracownicy)
        {
            System.out.println(p.id + " " + p.nazwisko + " " + p.stanowisko);
        }
    }
}

package lab6;

import java.util.Arrays;
import java.text.Collator;
import java.util.Locale;

class Ksiazka implements Comparable<Ksiazka> 
{
    String tytul;
    String autor;
    int rok;

    @Override
    public int compareTo(Ksiazka inna) 
    {
        Collator collator = Collator.getInstance(Locale.forLanguageTag("pl-PL"));

        return collator.compare(inna.tytul, this.tytul);
    }
}

public class Zadanie5_6 {
    public static void main(String[] args) {

        Ksiazka[] ksiazki = new Ksiazka[4];

        for (int i = 0; i < ksiazki.length; i++)
        {
            ksiazki[i] = new Ksiazka();
        }

        ksiazki[0].tytul = "Wiedzmin";
        ksiazki[0].autor = "Andrzej Sapkowski";
        ksiazki[0].rok = 1990;

        ksiazki[1].tytul = "Hobbit";
        ksiazki[1].autor = "J.R.R. Tolkien";
        ksiazki[1].rok = 1937;

        ksiazki[2].tytul = "Lalka";
        ksiazki[2].autor = "Boleslaw Prus";
        ksiazki[2].rok = 1890;

        ksiazki[3].tytul = "Dziady";
        ksiazki[3].autor = "Adam Mickiewicz";
        ksiazki[3].rok = 1823;
